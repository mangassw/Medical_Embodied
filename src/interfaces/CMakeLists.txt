  "msg/Fault.msg"
  "srv/Dock.srv"
  "srv/ChargeUntil.srv"

  "srv/SetConfig.srv"
  "action/Navigate.action"
  "action/Docking.action"
  "action/LLMInteraction.action"
  "action/CallNurse.action"
  DEPENDENCIES std_msgs
)

ament_export_dependencies(rosidl_default_runtime)

ament_package()
